import subprocess
import time

# Kill any processes running on port 8000 before starting a new one
try:
    subprocess.run(["killall", "uvicorn"], capture_output=True, check=False)
    print("Killed any existing uvicorn processes.")
except FileNotFoundError:
    print("killall command not found, skipping process termination.")

# Start uvicorn in a non-blocking way
process = subprocess.Popen(["uvicorn", "main:app", "--host", "0.0.0.0", "--port", "8000"], stdout=subprocess.PIPE, stderr=subprocess.PIPE)

print("FastAPI server started. Waiting for startup messages...")

# Read and print output for a few seconds to confirm startup
startup_messages = []
start_time = time.time()
while True:
    output = process.stdout.readline().decode().strip()
    if output:
        print(output)
        startup_messages.append(output)
    if "Application startup complete." in output or (time.time() - start_time > 10 and len(startup_messages) > 0):
        print("Server appears to have started.")
        break
    if process.poll() is not None: # Server terminated early
        print("Server terminated prematurely.")
        stderr_output = process.stderr.read().decode().strip()
        if stderr_output:
            print("Error during startup:\n" + stderr_output)
        break
    time.sleep(0.1)
